% Macros for styled dots
\def\ydot#1#2{
    \fill[black, opacity=0.15] (#1+0.15, #2+0.15) circle (0.8);
    \fill[samyellow, opacity=1] (#1, #2) circle (0.8);
}
\def\pdot#1#2{
    \fill[black, opacity=0.15] (#1+0.15, #2+0.15) circle (0.7);
    \fill[sampurple, opacity=1] (#1, #2) circle (0.7);
}
