{{/* Karnaugh map drawn as SVG. rows/cols: header labels, comma separated (Gray-code order as displayed) cells: values row by row, rows split by ";" (x = don't-care, blank = empty) idx: optional minterm numbers in the same layout groups: "r,c,h,w[,d]" per group, split by ";". Spans past the last row/column wrap around; ",d" draws the group dashed. Each group is inset more than the previous one, so loops sharing a cell nest. Styles live in assets/css/extended/kmap.css. */}} {{- $cs := 48 -}} {{- $ox := 62 -}} {{- $oy := 46 -}} {{- $rows := split (.Get "rows") "," -}} {{- $cols := split (.Get "cols") "," -}} {{- $nr := len $rows -}} {{- $nc := len $cols -}} {{- $gw := mul $nc $cs -}} {{- $gh := mul $nr $cs -}} {{- $W := add (add $ox $gw) 24 -}} {{- $H := add (add $oy $gh) 16 -}} {{- $cid := printf "kmap-clip-%d" .Ordinal -}} {{- $idx := split (.Get "idx" | default "") ";" -}}
