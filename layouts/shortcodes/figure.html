{{/* ---------- Reset on first figure per page ---------- */}} {{- if not (.Page.Scratch.Get "figureInit") -}} {{- .Page.Scratch.Delete "figureIDs" -}} {{- .Page.Scratch.Set "figureIDs" (dict) -}} {{- .Page.Scratch.Set "figureCount" 0 -}} {{- .Page.Scratch.Set "figureInit" true -}} {{- end -}} {{/* ---------- Get next number ---------- */}} {{- $figIDs := .Page.Scratch.Get "figureIDs" | default (dict) -}} {{- $count := .Page.Scratch.Get "figureCount" | default 0 -}} {{- $num := add $count 1 -}} {{- .Page.Scratch.Set "figureCount" $num -}} {{- $id := .Get "id" | default (printf "fig-%d" $num) -}} {{/* ---------- Register this figure ---------- */}} {{- $figIDs = merge $figIDs (dict (printf "%d" $num) $id) -}} {{- if .Get "name" -}} {{- $figIDs = merge $figIDs (dict (printf "%v" (.Get "name")) $id) -}} {{- end -}} {{- if .Get "id" -}} {{- $figIDs = merge $figIDs (dict (.Get "id") $id) -}} {{- end -}} {{- .Page.Scratch.Set "figureIDs" $figIDs -}} {{/* ---------- Other params ---------- */}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" | default "" -}} {{- $caption := .Get "caption" | default "" -}} {{- $width := .Get "width" | default "80%" -}}
