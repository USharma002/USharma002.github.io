Styles live in assets/css/extended/booth.css. */}} {{- $qs := split (.Get "q") "" -}} {{- $n := len $qs -}} {{- /* digit j = (bit to its right) - (bit j), with a 0 appended right of the LSB */ -}} {{- $digits := slice -}} {{- $qval := 0 -}} {{- $dval := 0 -}} {{- range $j, $b := $qs -}} {{- $right := 0 -}} {{- if lt $j (sub $n 1) }}{{ $right = int (index $qs (add $j 1)) }}{{ end -}} {{- $d := sub $right (int $b) -}} {{- $digits = $digits | append $d -}} {{- $qval = add (mul $qval 2) (int $b) -}} {{- $dval = add $dval (mul $d (int (math.Pow 2 (sub (sub $n 1) $j)))) -}} {{- end -}} {{- if eq (index $qs 0) "1" }}{{ $qval = sub $qval (int (math.Pow 2 $n)) }}{{ end -}} {{- $cw := 32 -}} {{- $x0 := 64 -}} {{- $W := add (add $x0 (mul (add $n 1) $cw)) 20 -}}
